**2. Add WiFi watchdog systemd timer + service in generate_systemd_services():**

Generate wifi-watchdog.timer:
```ini
[Unit]
Description=WiFi Watchdog Timer
After=touchscreen-check.service

[Timer]
OnBootSec=2min
OnUnitActiveSec=5min
AccuracySec=30s

[Install]
WantedBy=timers.target
```

Generate wifi-watchdog.service:
```ini
[Unit]
Description=WiFi Watchdog Check
After=network-online.target touchscreen-check.service
Wants=network-online.target

[Service]
Type=oneshot
ExecStart=${REPO_ROOT}/watchdog/wifi-watchdog.sh
StandardOutput=journal
StandardError=journal
```

Key details:
- Timer starts 2 minutes after boot (WiFi needs time to authenticate after cold boot)
- Timer fires every 5 minutes after that
- Service runs After touchscreen-check.service (coordination per NET-06)
- Type=oneshot because it runs once per check

**3. Add browser watchdog systemd service in generate_systemd_services():**

Generate browser-watchdog.service:
```ini
[Unit]
Description=Browser Idle Watchdog
After=graphical.target touchscreen-check.service
Wants=graphical.target

[Service]
Type=simple
ExecStart=${REPO_ROOT}/watchdog/browser-watchdog.sh
Restart=on-failure
RestartSec=10
StartLimitBurst=5
StartLimitIntervalSec=300
StandardOutput=journal
StandardError=journal
User=${USER}
Environment=DISPLAY=:0
```

Key details:
- Runs as user service with DISPLAY=:0 for xprintidle/xdotool access
- Restart=on-failure with rate limiting (5 restarts in 5 minutes max)
- After graphical.target so X11 is available

**4. Add scheduled reboot systemd timer + service (BRW-04):**

Load the reboot interval from config.json.template default (48 hours). Since bootstrap generates the timer at install time, read the value:
```bash
local reboot_hours
reboot_hours=$(jq -r '.browser.scheduled_reboot_interval_hours // 48' "${REPO_ROOT}/config.json" 2>/dev/null || echo "48")
```

Generate scheduled-reboot.timer:
```ini
[Unit]
Description=Scheduled Soft Reboot Timer

[Timer]
OnBootSec=${reboot_hours}h
AccuracySec=1h

[Install]
WantedBy=timers.target
```

Generate scheduled-reboot.service:
```ini
[Unit]
Description=Scheduled Soft Reboot

[Service]
Type=oneshot
ExecStart=/usr/bin/systemctl reboot
StandardOutput=journal
StandardError=journal
```

check_wifi_watchdog() {
  if [[ ! -f "${REPO_ROOT}/watchdog/wifi-watchdog.sh" ]]; then
    issues+=("wifi-watchdog.sh not found")
    return
  fi
  if [[ ! -x "${REPO_ROOT}/watchdog/wifi-watchdog.sh" ]]; then
    issues+=("wifi-watchdog.sh is not executable")
  fi
  if systemctl is-enabled wifi-watchdog.timer >/dev/null 2>&1; then
    log "wifi-watchdog.timer is enabled."
  else
    issues+=("wifi-watchdog.timer is not enabled; run bootstrap.sh")
  fi
}
